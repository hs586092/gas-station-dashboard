        async function loadCSV(url) {

            const response = await fetch(url);

            const text = await response.text();

            const lines = text.trim().split('\n');

            const headers = lines[0].split(',').map(h => h.trim());

            

            const data = [];

            for (let i = 1; i < lines.length; i++) {

                const values = lines[i].split(',');

                if (values.length === headers.length) {

                    const row = {};

                    headers.forEach((header, idx) => {

                        row[header] = values[idx].trim();

                    });

                    data.push(row);

                }

            }

            return data;

        }


        function analyzeData(data, fuelType) {

            console.log(`Analyzing ${fuelType} data:`, data.length, 'rows');

            

            // 만남의 광장 필터링 및 분석

            const mannamData = [];

            const competitorData = {};

            

            data.forEach(row => {

                const competitor = row['경쟁사'];

                const priceDiff = parseFloat(row['가격 차이']);

                const revenue = parseFloat(row['매출']);

                

                if (isNaN(priceDiff) || isNaN(revenue)) return;

                

                if (competitor === '만남의 광장') {

                    // 만남: -15 ~ +10 범위

                    if (priceDiff >= -15 && priceDiff <= 10) {

                        mannamData.push({priceDiff, revenue});

                    }

                } else {

                    if (!competitorData[competitor]) {

                        competitorData[competitor] = [];

                    }

                    competitorData[competitor].push({priceDiff, revenue});

                }

            });

            

            // 각 경쟁사별 IQR outlier 제거

            const processedData = [...mannamData];

            let outlierCount = 0;

            

            Object.keys(competitorData).forEach(competitor => {

                const compData = competitorData[competitor];

                const revenues = compData.map(d => d.revenue).sort((a, b) => a - b);

                

                const q1 = revenues[Math.floor(revenues.length * 0.25)];

                const q3 = revenues[Math.floor(revenues.length * 0.75)];

                const iqr = q3 - q1;

                const lowerBound = q1 - 1.5 * iqr;

                const upperBound = q3 + 1.5 * iqr;

                

                compData.forEach(d => {

                    if (d.revenue >= lowerBound && d.revenue <= upperBound) {

                        processedData.push(d);

                    } else {

                        outlierCount++;

                    }

                });

            });

            

            console.log(`Processed data: ${processedData.length} points (${outlierCount} outliers removed)`);

            

            // 회귀 분석

            const n = processedData.length;

            const x = processedData.map(d => d.priceDiff);

            const y = processedData.map(d => d.revenue);

            

            // Linear regression

            const sumX = x.reduce((a, b) => a + b, 0);

            const sumY = y.reduce((a, b) => a + b, 0);

            const sumXY = x.reduce((sum, xi, i) => sum + xi * y[i], 0);

            const sumX2 = x.reduce((sum, xi) => sum + xi * xi, 0);

            const sumY2 = y.reduce((sum, yi) => sum + yi * yi, 0);

            

            const meanX = sumX / n;

            const meanY = sumY / n;

            

            const slope = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);

            const intercept = meanY - slope * meanX;

            

            // R-squared

            const predictions = x.map(xi => slope * xi + intercept);

            const ssTot = y.reduce((sum, yi) => sum + Math.pow(yi - meanY, 2), 0);

            const ssRes = y.reduce((sum, yi, i) => sum + Math.pow(yi - predictions[i], 2), 0);

            const r2 = 1 - (ssRes / ssTot);

            

            // Adjusted R²

            const adjR2 = 1 - ((1 - r2) * (n - 1) / (n - 2));

            

            // Standard error

            const residuals = y.map((yi, i) => yi - predictions[i]);

            const rmse = Math.sqrt(ssRes / n);

            const stdError = Math.sqrt(ssRes / (n - 2));

            

            // Correlation

            const correlation = slope > 0 ? Math.sqrt(r2) : -Math.sqrt(r2);

            

            // F-statistic

            const msReg = (ssTot - ssRes) / 1;

            const msRes = ssRes / (n - 2);

            const fStat = msReg / msRes;

            

            // p-value (approximate)

            const tStat = slope / (stdError / Math.sqrt(sumX2 - sumX * sumX / n));

            

            // Polynomial degree 2

            const poly2 = fitPolynomial(x, y, 2);

            

            // Polynomial degree 3

            const poly3 = fitPolynomial(x, y, 3);

            

            // 최적 가격 찾기 (linear 기준)

            const optimalPrice = -intercept / slope;

            const maxRevenue = slope * optimalPrice + intercept;

            

            // 신뢰구간 (95%)

            const ciMargin = 1.96 * stdError;

            const ciLower = maxRevenue - ciMargin;

            const ciUpper = maxRevenue + ciMargin;

            

            // 잔차 분석

            const residualMean = residuals.reduce((a, b) => a + b, 0) / n;

            const residualStd = Math.sqrt(residuals.reduce((sum, r) => sum + Math.pow(r - residualMean, 2), 0) / n);

            const skewness = calculateSkewness(residuals);

            const kurtosis = calculateKurtosis(residuals);

            

            // Durbin-Watson

            const dw = calculateDurbinWatson(residuals);

            

            // Elasticity

            const elasticity = slope * (meanX / meanY);

            

            return {

                n,

                outlierCount,

                optimalPrice,

                maxRevenue,

                slope,

                intercept,

                r2,

                adjR2,

                rmse,

                stdError,

                correlation,

                fStat,

                tStat,

                ciLower,

                ciUpper,

                ciMargin,

                residualMean,

                residualStd,

                skewness,

                kurtosis,

                dw,

                elasticity,

                poly2,

                poly3

            };

        }

        

        function fitPolynomial(x, y, degree) {

            const n = x.length;

            

            // Create design matrix

            const X = [];

            for (let i = 0; i < n; i++) {

                const row = [];

                for (let d = 0; d <= degree; d++) {

                    row.push(Math.pow(x[i], d));

                }

                X.push(row);

            }

            

            // Normal equations: (X^T X) β = X^T y

            const XtX = multiplyMatrices(transpose(X), X);

            const Xty = multiplyMatrixVector(transpose(X), y);

            

            // Solve using Gaussian elimination

            const coeffs = gaussianElimination(XtX, Xty);

            

            // Calculate R²

            const predictions = x.map((xi, i) => {

                let pred = 0;

                for (let d = 0; d <= degree; d++) {

                    pred += coeffs[d] * Math.pow(xi, d);

                }

                return pred;

            });

            

            const meanY = y.reduce((a, b) => a + b, 0) / n;

            const ssTot = y.reduce((sum, yi) => sum + Math.pow(yi - meanY, 2), 0);

            const ssRes = y.reduce((sum, yi, i) => sum + Math.pow(yi - predictions[i], 2), 0);

            const r2 = 1 - (ssRes / ssTot);

            const adjR2 = 1 - ((1 - r2) * (n - 1) / (n - degree - 1));

            const rmse = Math.sqrt(ssRes / n);

            

            return {coeffs, r2, adjR2, rmse};

        }

        

        function transpose(matrix) {

            return matrix[0].map((_, i) => matrix.map(row => row[i]));

        }

        

        function multiplyMatrices(a, b) {

            const result = [];

            for (let i = 0; i < a.length; i++) {

                result[i] = [];

                for (let j = 0; j < b[0].length; j++) {

                    let sum = 0;

                    for (let k = 0; k < a[0].length; k++) {

                        sum += a[i][k] * b[k][j];

                    }

                    result[i][j] = sum;

                }

            }

            return result;

        }

        

        function multiplyMatrixVector(matrix, vector) {

            return matrix.map(row => 

                row.reduce((sum, val, i) => sum + val * vector[i], 0)

            );

        }

        

        function gaussianElimination(A, b) {

            const n = A.length;

            const aug = A.map((row, i) => [...row, b[i]]);

            

            // Forward elimination

            for (let i = 0; i < n; i++) {

                // Partial pivoting

                let maxRow = i;

                for (let k = i + 1; k < n; k++) {

                    if (Math.abs(aug[k][i]) > Math.abs(aug[maxRow][i])) {

                        maxRow = k;

                    }

                }

                [aug[i], aug[maxRow]] = [aug[maxRow], aug[i]];

                

                // Eliminate column

                for (let k = i + 1; k < n; k++) {

                    const factor = aug[k][i] / aug[i][i];

                    for (let j = i; j <= n; j++) {

                        aug[k][j] -= factor * aug[i][j];

                    }

                }

            }

            

            // Back substitution

            const x = new Array(n);

            for (let i = n - 1; i >= 0; i--) {

                x[i] = aug[i][n];

                for (let j = i + 1; j < n; j++) {

                    x[i] -= aug[i][j] * x[j];

                }

                x[i] /= aug[i][i];

            }

            

            return x;

        }

        

        function calculateSkewness(values) {

            const n = values.length;

            const mean = values.reduce((a, b) => a + b, 0) / n;

            const std = Math.sqrt(values.reduce((sum, v) => sum + Math.pow(v - mean, 2), 0) / n);

            const skew = values.reduce((sum, v) => sum + Math.pow((v - mean) / std, 3), 0) / n;

            return skew;

        }

        

        function calculateKurtosis(values) {

            const n = values.length;

            const mean = values.reduce((a, b) => a + b, 0) / n;

            const std = Math.sqrt(values.reduce((sum, v) => sum + Math.pow(v - mean, 2), 0) / n);

            const kurt = values.reduce((sum, v) => sum + Math.pow((v - mean) / std, 4), 0) / n;

            return kurt - 3; // Excess kurtosis

        }

        

        function calculateDurbinWatson(residuals) {

            let sumDiff = 0;

            let sumSquared = 0;

            

            for (let i = 1; i < residuals.length; i++) {

                sumDiff += Math.pow(residuals[i] - residuals[i-1], 2);

            }

            

            for (let i = 0; i < residuals.length; i++) {

                sumSquared += Math.pow(residuals[i], 2);

            }

            

            return sumDiff / sumSquared;

        }
